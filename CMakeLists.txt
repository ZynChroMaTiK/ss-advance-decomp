    target_include_directories(ssa_vulkan_backend PUBLIC include platform/win32)

    if(SSA_VULKAN)
        find_package(Vulkan QUIET)
        find_program(SSA_GLSLANG NAMES glslangValidator HINTS "$ENV{VULKAN_SDK}/Bin")
        if(Vulkan_FOUND AND SSA_GLSLANG)
            file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/shaders")
            foreach(stage vert frag)
                set(header "${CMAKE_CURRENT_BINARY_DIR}/shaders/ssa_scene_${stage}.h")
                add_custom_command(OUTPUT "${header}"
                    COMMAND "${SSA_GLSLANG}" -V --target-env vulkan1.0 --vn "ssa_scene_${stage}"
                        -o "${header}" "${CMAKE_CURRENT_SOURCE_DIR}/platform/vulkan/scene.${stage}"
                    DEPENDS "platform/vulkan/scene.${stage}" VERBATIM)
                target_sources(ssa_vulkan_backend PRIVATE "${header}")
            endforeach()
            target_include_directories(ssa_vulkan_backend PRIVATE ${Vulkan_INCLUDE_DIRS} "${CMAKE_CURRENT_BINARY_DIR}/shaders")
            target_compile_definitions(ssa_vulkan_backend PRIVATE SSA_HAVE_VULKAN)
        else()
            message(STATUS "Vulkan SDK/shader compiler not found; legacy renderers remain available")
        endif()
    endif()
    if(MSVC)
        target_compile_options(ssa_vulkan_backend PRIVATE /W4 /WX)
    endif()
endif()
